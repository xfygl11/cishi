#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
饭太硬 TV 数据抓取 - 完整流程
"""

import subprocess
import base64
import zlib
import json
import re
import os

# 第1步: 使用 curl 下载原始数据
def download_config(url="http://www.饭太硬.net/tv"):
    """下载配置数据"""
    result = subprocess.run(
        ['curl', '-s', '--max-time', '30', '-o', '/tmp/tv_data.bin', url],
        capture_output=True, text=True
    )
    if result.returncode != 0 or not os.path.exists('/tmp/tv_data.bin'):
        raise Exception(f"下载失败: {result.stderr}")
    
    file_size = os.path.getsize('/tmp/tv_data.bin')
    print(f"[OK] 下载成功，文件大小: {file_size} 字节")
    return '/tmp/tv_data.bin'

# 第2步: Base64 解码
def decode_base64(filepath):
    """解码 Base64 数据"""
    with open(filepath, 'rb') as f:
        data = f.read()
    
    text = data.decode('utf-8', errors='ignore')
    
    # 正则匹配 Base64 字符串
    b64_pattern = r'([A-Za-z0-9+/]{200,}={0,2})'
    matches = re.findall(b64_pattern, text)
    
    for match in matches:
        try:
            decoded = base64.b64decode(match)
            content = decoded.decode('utf-8', errors='ignore')
            if '"sites"' in content:
                print(f"[OK] Base64 解码成功")
                return content
        except Exception as e:
            continue
    
    raise Exception("未找到有效的 JSON 数据")

# 第3步: zlib 解压缩（如果数据被压缩）
def decompress_if_needed(data):
    """如果数据被 zlib 压缩则解压"""
    try:
        decompressed = zlib.decompress(data)
        print(f"[OK] zlib 解压缩成功")
        return decompressed.decode('utf-8')
    except:
        return data

# 第4步: 解析 JSON
def parse_json(content):
    """解析 JSON 配置"""
    try:
        config = json.loads(content)
        print(f"[OK] JSON 解析成功")
        return config
    except json.JSONDecodeError as e:
        print(f"[WARN] JSON 解析失败: {e}")
        print(f"     尝试手动解析...")
        return manual_parse(content)

def manual_parse(content):
    """手动解析 JSON（处理格式问题）"""
    # 移除注释行
    lines = content.split('\n')
    clean_lines = [l for l in lines if not l.strip().startswith('//')]
    content = '\n'.join(clean_lines)
    
    # 提取 sites 数组
    sites_match = re.search(r'"sites":\[(.*?)\]', content, re.DOTALL)
    if sites_match:
        sites_str = '[' + sites_match.group(1) + ']'
        # 逐行解析每个站点
        sites = []
        current = ''
        for line in sites_str.split('\n'):
            line = line.strip().rstrip(',')
            if line.startswith('{') and line.endswith('}'):
                try:
                    sites.append(json.loads(line))
                except:
                    pass
            elif line.startswith('{'):
                current = line
            elif current:
                current += ' ' + line
                if '}' in current:
                    try:
                        sites.append(json.loads(current))
                    except:
                        pass
                    current = ''
        
        print(f"[OK] 手动解析成功，找到 {len(sites)} 个站点")
        return {'sites': sites}
    
    return {}

# 主流程
def main():
    print("=" * 70)
    print("  饭太硬 TV 数据抓取 - 完整流程")
    print("=" * 70)
    
    # 1. 下载
    filepath = download_config()
    
    # 2. 解码
    content = decode_base64(filepath)
    
    # 3. 解析
    config = parse_json(content)
    
    # 4. 输出

        print(f"\n[INFO] 共找到 {len(config['sites'])} 个站点")
        for i, site in enumerate(config['sites'][:5]):
            print(f"  [{i+1}] {site.get('name', '未知')}")
    
    print("\n" + "=" * 70)
    print("  流程完成！")
    print("=" * 70)

if __name__ == '__main__':
    main()

def parse_sites_config(content):
    """解析站点配置"""
    # 移除注释
    lines = content.split('\n')
    clean_lines = [l for l in lines if not l.strip().startswith('//')]
    content = '\n'.join(clean_lines)
    
    sites = []
    current_site = ''
    in_sites = False
    
    for line in content.split('\n'):
        stripped = line.strip().rstrip(',')
        
        if '"sites":[' in line:
            in_sites = True
            after = line.split('[', 1)[1]
            if '{' in after:
                si = after.index('{')
                rem = after[si:]
                if '}' in rem:
                    ei = rem.index('}') + 1
                    obj = rem[:ei].rstrip(',')
                    if obj: sites.append(obj)
                    rest = rem[ei:].strip().rstrip(',')
                    if rest and rest != ',' and rest.startswith('{'):
                        sites.append(rest)
            continue
        
        if in_sites:
            if ']' in line:
                before = line.split(']', 1)[0]
                bstripped = before.strip().rstrip(',')
                if bstripped.startswith('{'):
                    sites.append(bsstripped)
                in_sites = False
                continue
            
            if stripped.startswith('{') and stripped.endswith('}'):
                sites.append(stripped)
            elif stripped.startswith('{'):
                current_site = stripped.rstrip(',')
            elif current_site:
                current_site = current_site.rstrip(',') + ' ' + stripped.rstrip(',')
                if '}' in current_site:
                    sites.append(current_site)
                    current_site = ''
    
    # 解析 JSON
    parsed_sites = []
    for s in sites:
        try:
            parsed_sites.append(json.loads(s))
        except json.JSONDecodeError:
            pass
    
    return parsed_sites

def find_site(sites, keyword):
    """根据关键词查找站点"""
    results = []
    for site in sites:
        name = site.get('name', '')
        key = site.get('key', '')

def categorize_sites(sites):
    """按内容类型对站点进行分类"""
    categories = {
        '影视点播': [],
        '动漫': [],
        '直播/体育': [],
        '网盘/搜索': [],
        '教育/其他': []
    }
    
    for site in sites:
        name = site.get('name', '')
        
        if any(k in name for k in ['不卡', '秒播', '多线', '磁力']):
            categories['影视点播'].append(site)
        elif any(k in name for k in ['动漫', '巴士', '日本', '咕咕']):
            categories['动漫'].append(site)
        elif any(k in name for k in ['看球', '直播', 'MV', '虎牙', '斗鱼']):
            categories['直播/体育'].append(site)
        elif any(k in name for k in ['搜', '盘', '聚剧', '聚盘', '云盘']):
            categories['网盘/搜索'].append(site)
        else:
            categories['教育/其他'].append(site)
    
    return categories

5.2 搜索能力统计

def search_capability(sites):
    """统计站点的搜索能力"""
    stats = {
        'fast_and_normal': [],  # 快速+普通
        'normal_only': [],       # 仅普通
        'none': []               # 不可搜索
    }
    
    for site in sites:
        searchable = site.get('searchable', 0)
        quick_search = site.get('quickSearch', 0)
        
        if searchable and quick_search:
            stats['fast_and_normal'].append(site)
        elif searchable:
            stats['normal_only'].append(site)
        else:
            stats['none'].append(site)
    
    return stats

5.3 提取云盘配置站点

def get_cloud_drive_sites(sites):
    """获取需要云盘配置的站点"""
    cloud_sites = []
    for site in sites:
        ext = site.get('ext', {})
        if isinstance(ext, dict) and 'Cloud-drive' in ext:
            cloud_sites.append({
                'name': site['name'],
                'key': site['key'],
                'cloud_drive': ext['Cloud-drive']
            })
    return cloud_sites

5.4 提取外部 URL 依赖

def extract_external_urls(sites):
    """提取站点的外部 URL 依赖"""
    urls = {
        'siteUrls': [],
        'jsonConfigs': [],
        'jsScripts': []
    }
    
    for site in sites:
        ext = site.get('ext', '')
        api = site.get('api', '')
        
        # 检查 siteUrl
        if isinstance(ext, dict) and 'siteUrl' in ext:
            urls['siteUrls'].append({
                'name': site['name'],
                'url': ext['siteUrl']
            })
        
        # 检查 json 配置
        if isinstance(ext, dict) and 'json' in ext:
            urls['jsonConfigs'].append({
                'name': site['name'],
                'url': ext['json']
            })
        
        # 检查 js 脚本
        if 'drpy2.min.js' in api:
            urls['jsScripts'].append({
                'name': site['name'],
                'api': api,
                'ext': ext if isinstance(ext, str) else ''
            })
    
    return urls

    print(f"    解码成功 ({len(content)} 字符)")
    
    # 解析
    print("[3/3] 正在解析站点...")
    sites = parse_sites(content)
    print(f"    解析成功: {len(sites)} 个站点")
    
    # 输出
    if args.wanou:
        wanou_sites = find_site(sites, '玩偶')
        if wanou_sites:
            print("\n" + "=" * 70)
            print("  玩偶哥哥 详细信息")
            print("=" * 70)
            print(json.dumps(wanou_sites[0], ensure_ascii=False, indent=2))
    elif args.site:
        matched = find_site(sites, args.site)
        if matched:
            print(f"\n找到 {len(matched)} 个匹配站点:")
            display_sites(matched, show_all=True)
        else:
            print(f"未找到包含 '{args.site}' 的站点")
    elif args.json:
        output = {
            'total': len(sites),
            'sites': sites
        }
        if args.output:
            with open(args.output, 'w', encoding='utf-8') as f:
                json.dump(output, f, ensure_ascii=False, indent=2)
            print(f"\nJSON 已保存到: {args.output}")
        else:
            print(json.dumps(output, ensure_ascii=False, indent=2))
    else:
        print(f"\n共 {len(sites)} 个站点:")
        display_sites(sites, show_all=True)
    
    print("=" * 70)
    print("  完成！")
    print("=" * 70)

if __name__ == '__main__':
    main()
